    ```ts
    import type { UpdateUserRequest, Permission } from '../../../shared/types/user';

    export interface UpdateUserErrors {
      username?: FieldError;
      password?: FieldError;
      role?: FieldError;
      permissions?: FieldError;
    }

    /**
     * Edit-form validator. Any field may be omitted (means "no change").
     * Empty-string password is treated as "no change" to match the UI contract.
     */
    export function validateUpdateUser(input: UpdateUserRequest): UpdateUserErrors {
      const errs: UpdateUserErrors = {};
      if (input.username !== undefined) {
        const u = validateUsername(input.username);
        if (u) errs.username = u;
      }
      if (input.password !== undefined && input.password !== '') {
        const p = validatePassword(input.password);
        if (p) errs.password = p;
      }
      if (input.role !== undefined) {
        const r = validateRole(input.role as string);
        if (r) errs.role = r;
      }
      if (input.permissions !== undefined) {
        const perms = validatePermissions(input.permissions as Permission[]);
        if (perms) errs.permissions = perms;
      }
      return errs;
    }
    ```

    ```ts
    const { user: currentUser } = useAuth();
    const isSelf = (r: UserListItem) => !!currentUser && currentUser.id === r.id;

    // Edit modal state
    const [editTarget, setEditTarget] = useState<UserListItem | null>(null);
    const [editForm, setEditForm]     = useState<{ password: string; role: Role; permissions: Permission[] }>({
      password: '', role: 'staff', permissions: [],
    });
    const [editErrs, setEditErrs]     = useState<UpdateUserErrors>({});
    const [editSaving, setEditSaving] = useState(false);
    const [editError, setEditError]   = useState<string | null>(null);

    const openEdit = (r: UserListItem) => {
      setEditTarget(r);
      setEditForm({ password: '', role: r.role, permissions: r.permissions });
      setEditErrs({});
      setEditError(null);
    };

    const onEditSave = async () => {
      if (!editTarget) return;
      const patch: UpdateUserRequest = { role: editForm.role, permissions: editForm.permissions };
      if (editForm.password) patch.password = editForm.password;
      const v = validateUpdateUser(patch);
      setEditErrs(v);
      if (Object.keys(v).length) return;
      setEditSaving(true); setEditError(null);
      try {
        const updated = await userService.update(editTarget.id, patch);
        setRows((prev) => prev ? prev.map((u) => (u.id === updated.id ? updated : u)) : prev);
        setEditTarget(null);
      } catch (e: any) {
        const code = e?.response?.data?.error;
        if (code === 'username_taken')        setEditError('That username is already taken.');
        else if (code === 'invalid_password') setEditError('Password must be at least 8 characters.');
        else if (code === 'invalid_role')     setEditError('Role must be Admin or Staff.');
        else if (code === 'invalid_permissions') setEditError('Select at least one permission.');
        else if (code === 'user_not_found')   setEditError('This user no longer exists — refreshing list.');
        else                                  setEditError('Could not save changes. Try again.');
      } finally {
        setEditSaving(false);
      }
    };

    // Deactivate confirm state
    const [deactivateTarget, setDeactivateTarget] = useState<UserListItem | null>(null);
    const [deactivating, setDeactivating]         = useState(false);

    const openDeactivate = (r: UserListItem) => {
      if (isSelf(r)) return;                   // client-side self-lockout guard (T-02-14)
      setDeactivateTarget(r);
    };

    const onDeactivateConfirm = async () => {
      if (!deactivateTarget) return;
      setDeactivating(true);
      try {
        const updated = await userService.deactivate(deactivateTarget.id);
        setRows((prev) => prev ? prev.map((u) => (u.id === updated.id ? updated : u)) : prev);
        setDeactivateTarget(null);
      } catch (e: any) {
        const code = e?.response?.data?.error;
        if (code === 'self_lockout_forbidden') {
          Alert.alert('Action blocked', 'You cannot deactivate your own account.');
        } else if (code === 'user_not_found') {
          Alert.alert('Not found', 'This user no longer exists — refreshing list.');
          await load();
        } else {
          Alert.alert('Error', 'Could not deactivate user. Try again.');
        }
        setDeactivateTarget(null);
      } finally {
        setDeactivating(false);
      }
    };
    ```
